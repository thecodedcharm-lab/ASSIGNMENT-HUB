16-B. Mini Assignment 2: Classroom Attendance Tracker
const attendance = [
    ["Jenil", ["P", "P", "A", "P", "P"]],
    ["Mahir", ["P", "A", "P", "P", "A"]],
    ["Yashvi", ["P", "P", "P", "P", "P"]],
    ["Arjun", ["A", "A", "P", "P", "A"]]
];

let highestAttendance = 0;
let highestStudent = "";

for (let i = 0; i < attendance.length; i++) {
    const name = attendance[i][0];
    const records = attendance[i][1];

    let present = 0;
    let absent = 0;

    for (let j = 0; j < records.length; j++) {
        if (records[j] === "P") {
            present++;
        } else {
            absent++;
        }
    }

    const percentage = (present / records.length) * 100;

    let status;

    if (percentage >= 75) {
        status = "Good";
    } else {
        status = "Low Attendance";
    }

    console.log("Name:", name);
    console.log("Present:", present);
    console.log("Absent:", absent);
    console.log("Attendance:", percentage + "%");
    console.log("Status:", status);

    if (percentage > highestAttendance) {
        highestAttendance = percentage;
        highestStudent = name;
    }
}

console.log("Highest Attendance:", highestStudent);

16-C. Mini Assignment 3: Movie Rating Analyzer
const movies = [
    ["Inception", [9, 8, 9, 10, 8]],
    ["Avatar", [7, 8, 7, 6, 8]],
    ["Interstellar", [10, 9, 9, 10, 9]],
    ["Joker", [8, 7, 8, 9, 8]]
];

let highestAverage = 0;
let highestMovie = "";

for (let i = 0; i < movies.length; i++) {
    const movieName = movies[i][0];
    const ratings = movies[i][1];

    let total = 0;

    for (let j = 0; j < ratings.length; j++) {
        total += ratings[j];
    }

    const average = total / ratings.length;

    let verdict;

    if (average >= 9) {
        verdict = "Excellent";
    } else if (average >= 7) {
        verdict = "Good";
    } else {
        verdict = "Average";
    }

    console.log("Movie:", movieName);
    console.log("Ratings:", ratings.join(" "));
    console.log("Average Rating:", average);
    console.log("Verdict:", verdict);

    if (average > highestAverage) {
        highestAverage = average;
        highestMovie = movieName;
    }
}

console.log("Highest Rated Movie:", highestMovie);
console.log("Highest Average:", highestAverage);

16-D. Mini Assignment 4: Temperature of the Week
const temperatures = [
    ["Monday", 32],
    ["Tuesday", 35],
    ["Wednesday", 30],
    ["Thursday", 28],
    ["Friday", 33],
    ["Saturday", 36],
    ["Sunday", 31]
];

let highestTemperature = temperatures[0][1];
let lowestTemperature = temperatures[0][1];

let highestDay = temperatures[0][0];
let lowestDay = temperatures[0][0];

let total = 0;

for (let i = 0; i < temperatures.length; i++) {
    const day = temperatures[i][0];
    const temperature = temperatures[i][1];

    console.log(day + ":", temperature);

    total += temperature;

    if (temperature > highestTemperature) {
        highestTemperature = temperature;
        highestDay = day;
    }

    if (temperature < lowestTemperature) {
        lowestTemperature = temperature;
        lowestDay = day;
    }

    if (temperature >= 35) {
        console.log("Heat Alert on", day);
    }
}

const average = total / temperatures.length;

console.log("Highest Temperature:", highestTemperature, "on", highestDay);
console.log("Lowest Temperature:", lowestTemperature, "on", lowestDay);
console.log("Average Temperature:", average);
